<script>
(function () {
  var ORIGIN = "https://boardofprojectstewardship.com";
  var f = document.getElementById("bops-magnolia-frame");
  if (!f) return;
  window.addEventListener("message", function (e) {
    if (e.origin !== ORIGIN || e.source !== f.contentWindow || !e.data || e.data.type !== "bops-magnolia:resize") return;
    f.style.height = Math.max(480, Math.min(4000, Number(e.data.height) || 820)) + "px";
  });
  f.addEventListener("load", function () { f.contentWindow.postMessage("bops-magnolia:ping", ORIGIN); });
})();
</script>
